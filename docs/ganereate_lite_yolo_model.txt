1. Python latest version install
sudo apt update
sudo apt install -y python3.15

2. virtual environment setting
python -m venv venv

3. virtual environment execute
source venv/bin/activate

4. Pytorch and yolo install
pip install --upgrade pip
pip install torch torchvision torchaudio
pip install -U ultralytics

5. generate lite yolo model file (./yolo26n_executorch_model/model.pte)
yolo export model=yolo26n.pt format=executorch imgsz=640



* pytorch version
pip show torch

* yolo version
pip show ultralytics
